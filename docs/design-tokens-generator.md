# Design Tokens 生成・同期ガイド

## 概要

`design-tokens.json` は唯一の真実の源（Single Source of Truth）です。このドキュメントは、Design Tokens JSON から以下を自動生成する方法を説明します：

1. **CSS 変数ファイル** (`src/styles/tokens.css`)
2. **Tailwind Config 拡張** (`apps/web/tailwind.config.ts`)
3. **React Constants** (`src/constants/design-tokens.ts`)
4. **型定義** (`src/types/tokens.d.ts`)

---

## 1. Design Tokens JSON 構造

### 色（Colors）

```json
{
  "colors": {
    "primary": {
      "50": { "value": "#F0F7FF", "description": "..." },
      "100": { "value": "#E0EFFF", "description": "..." },
      "500": { "value": "#0066CC", "description": "..." },
      ...
    },
    "semantic": {
      "success": {
        "500": { "value": "#10B981", "description": "..." },
        ...
      }
    },
    "neutral": { ... }
  }
}
```

### タイポグラフィ（Typography）

```json
{
  "typography": {
    "display1": {
      "fontSize": "48px",
      "fontWeight": "700",
      "lineHeight": "1.1",
      "letterSpacing": "-1px",
      "description": "..."
    },
    ...
  }
}
```

### スペーシング・ボーダーラディウス・シャドウ等

```json
{
  "spacing": { ... },
  "borderRadius": { ... },
  "shadow": { ... },
  "transition": { ... },
  "zIndex": { ... }
}
```

## 2. CSS 変数生成スクリプト

### ファイル: `scripts/generate-design-tokens.js`

```javascript
#!/usr/bin/env node
/**
 * Design Tokens JSON → CSS 変数・Tailwind config・React constants に変換
 *
 * 使用方法:
 *   node scripts/generate-design-tokens.js
 *
 * 生成ファイル:
 *   - src/styles/tokens.css (CSS 変数)
 *   - apps/web/tailwind.config.generated.ts (Tailwind theme)
 *   - src/constants/design-tokens.ts (React constants)
 *   - src/types/tokens.d.ts (TypeScript 型定義)
 */

const fs = require('fs');
const path = require('path');

// Load design tokens
const tokensPath = path.join(__dirname, '../docs/design-tokens.json');
const tokens = JSON.parse(fs.readFileSync(tokensPath, 'utf-8'));

// ============================================================================
// 1. CSS Variables Generator
// ============================================================================

function generateCSSVariables() {
  let css = `/**
 * Generated Design Tokens - CSS Variables
 * DO NOT EDIT MANUALLY - Generated by scripts/generate-design-tokens.js
 * Source: docs/design-tokens.json
 */

:root {
  /* Colors - Primary */
`;

  // Primary colors
  Object.entries(tokens.colors.primary).forEach(([key, value]) => {
    css += `  --color-primary-${key}: ${value.value};\n`;
  });

  css += `\n  /* Colors - Semantic */\n`;
  // Semantic colors
  Object.entries(tokens.colors.semantic).forEach(([semanticKey, semanticValue]) => {
    Object.entries(semanticValue).forEach(([key, value]) => {
      if (typeof value === 'object' && value.value) {
        css += `  --color-${semanticKey}-${key}: ${value.value};\n`;
      }
    });
  });

  css += `\n  /* Colors - Neutral (Grayscale) */\n`;
  // Neutral colors
  Object.entries(tokens.colors.neutral).forEach(([key, value]) => {
    css += `  --color-neutral-${key}: ${value.value};\n`;
  });

  css += `\n  /* Typography */\n`;
  // Typography
  Object.entries(tokens.typography).forEach(([key, value]) => {
    css += `  --font-${key}-size: ${value.fontSize};\n`;
    css += `  --font-${key}-weight: ${value.fontWeight};\n`;
    css += `  --font-${key}-line-height: ${value.lineHeight};\n`;
    css += `  --font-${key}-letter-spacing: ${value.letterSpacing};\n`;
  });

  css += `\n  /* Spacing */\n`;
  Object.entries(tokens.spacing).forEach(([key, value]) => {
    css += `  --spacing-${key}: ${value.value};\n`;
  });

  css += `\n  /* Border Radius */\n`;
  Object.entries(tokens.borderRadius).forEach(([key, value]) => {
    css += `  --radius-${key}: ${value.value};\n`;
  });

  css += `\n  /* Shadow / Elevation */\n`;
  Object.entries(tokens.shadow).forEach(([key, value]) => {
    css += `  --shadow-${key}: ${value.value};\n`;
  });

  css += `\n  /* Z-Index */\n`;
  Object.entries(tokens.zIndex).forEach(([key, value]) => {
    css += `  --z-${key}: ${value.value};\n`;
  });

  css += `\n  /* Transition */\n`;
  Object.entries(tokens.transition).forEach(([key, value]) => {
    css += `  --duration-${key}: ${value.value};\n`;
  });

  css += `\n  /* Dark Mode Overrides */\n`;
  css += `  --bg-primary: #FFFFFF;\n`;
  css += `  --bg-secondary: #F5F5F5;\n`;
  css += `  --text-primary: #212121;\n`;
  css += `  --text-secondary: #757575;\n`;
  css += `  --border-primary: #E0E0E0;\n`;
  css += `}\n`;

  // Dark mode
  css += `\n@media (prefers-color-scheme: dark) {\n`;
  css += `  :root {\n`;
  css += `    --bg-primary: #121212;\n`;
  css += `    --bg-secondary: #1E1E1E;\n`;
  css += `    --text-primary: #FFFFFF;\n`;
  css += `    --text-secondary: #BDBDBD;\n`;
  css += `    --border-primary: #424242;\n`;
  css += `  }\n`;
  css += `}\n`;

  // Manual dark mode class
  css += `\n.dark {\n`;
  css += `  --bg-primary: #121212;\n`;
  css += `  --bg-secondary: #1E1E1E;\n`;
  css += `  --text-primary: #FFFFFF;\n`;
  css += `  --text-secondary: #BDBDBD;\n`;
  css += `  --border-primary: #424242;\n`;
  css += `}\n`;

  return css;
}

// ============================================================================
// 2. Tailwind Config Generator
// ============================================================================

function generateTailwindConfig() {
  let config = `/**
 * Tailwind CSS Configuration - Generated from Design Tokens
 * DO NOT EDIT MANUALLY - Generated by scripts/generate-design-tokens.js
 */

export default {
  theme: {
    extend: {
      colors: {
`;

  // Primary
  config += `        primary: {\n`;
  Object.entries(tokens.colors.primary).forEach(([key, value]) => {
    config += `          ${key}: '${value.value}',\n`;
  });
  config += `        },\n`;

  // Semantic
  Object.entries(tokens.colors.semantic).forEach(([semanticKey, semanticValue]) => {
    config += `        ${semanticKey}: {\n`;
    Object.entries(semanticValue).forEach(([key, value]) => {
      if (typeof value === 'object' && value.value) {
        config += `          ${key}: '${value.value}',\n`;
      }
    });
    config += `        },\n`;
  });

  // Neutral
  config += `        neutral: {\n`;
  Object.entries(tokens.colors.neutral).forEach(([key, value]) => {
    config += `          ${key}: '${value.value}',\n`;
  });
  config += `        },\n`;

  config += `      },\n`;

  // Typography
  config += `      fontSize: {\n`;
  Object.entries(tokens.typography).forEach(([key, value]) => {
    config += `        '${key}': ['${value.fontSize}', { lineHeight: '${value.lineHeight}', fontWeight: '${value.fontWeight}', letterSpacing: '${value.letterSpacing}' }],\n`;
  });
  config += `      },\n`;

  // Spacing
  config += `      spacing: {\n`;
  Object.entries(tokens.spacing).forEach(([key, value]) => {
    config += `        '${key}': '${value.value}',\n`;
  });
  config += `      },\n`;

  // Border Radius
  config += `      borderRadius: {\n`;
  Object.entries(tokens.borderRadius).forEach(([key, value]) => {
    config += `        '${key}': '${value.value}',\n`;
  });
  config += `      },\n`;

  // Shadow
  config += `      boxShadow: {\n`;
  Object.entries(tokens.shadow).forEach(([key, value]) => {
    config += `        '${key}': '${value.value}',\n`;
  });
  config += `      },\n`;

  // Z-Index
  config += `      zIndex: {\n`;
  Object.entries(tokens.zIndex).forEach(([key, value]) => {
    config += `        '${key}': ${value.value},\n`;
  });
  config += `      },\n`;

  // Transition Duration
  config += `      transitionDuration: {\n`;
  Object.entries(tokens.transition).forEach(([key, value]) => {
    config += `        '${key}': '${value.value}',\n`;
  });
  config += `      },\n`;

  // Breakpoints
  config += `      screens: {\n`;
  Object.entries(tokens.breakpoints).forEach(([key, value]) => {
    if (key !== 'mobile') {
      config += `        '${key}': '${value.value}',\n`;
    }
  });
  config += `      },\n`;

  config += `    },\n`;
  config += `  },\n`;
  config += `};\n`;

  return config;
}

// ============================================================================
// 4. TypeScript Type Definitions Generator
// ============================================================================

function generateTypeDefinitions() {
  let ts = `/**
 * Design Tokens Type Definitions
 * DO NOT EDIT MANUALLY - Generated by scripts/generate-design-tokens.js
 */

export type ColorKey = keyof typeof COLORS;
export type ColorLevel = keyof typeof COLORS.primary;
export type SemanticColorKey = 'success' | 'warning' | 'error' | 'info';
export type TypographyKey = keyof typeof TYPOGRAPHY;
export type SpacingKey = keyof typeof SPACING;
export type BorderRadiusKey = keyof typeof BORDER_RADIUS;
export type ShadowKey = keyof typeof SHADOW;
export type ZIndexKey = keyof typeof Z_INDEX;

// Component Variants
export type ButtonVariant = 'primary' | 'secondary' | 'danger' | 'ghost';
export type ButtonSize = 'xs' | 'sm' | 'md' | 'lg';
export type ButtonStyle = 'solid' | 'outline' | 'ghost';

// Responsive Breakpoints
export type Breakpoint = 'xs' | 'sm' | 'md' | 'lg' | 'xl' | '2xl';

// Dark Mode
export type ColorScheme = 'light' | 'dark' | 'auto';
`;

  return ts;
}

// ============================================================================
// Main Execution
// ============================================================================

function main() {
  console.log('🎨 Generating Design Tokens...\n');

  // Create output directories
  const cssDir = path.join(__dirname, '../src/styles');
  const tsDir = path.join(__dirname, '../src/constants');
  const typesDir = path.join(__dirname, '../src/types');

  [cssDir, tsDir, typesDir].forEach((dir) => {
    if (!fs.existsSync(dir)) {
      fs.mkdirSync(dir, { recursive: true });
    }
  });

  // Generate CSS Variables
  console.log('📝 Generating CSS Variables...');
  const css = generateCSSVariables();
  const cssPath = path.join(cssDir, 'tokens.css');
  fs.writeFileSync(cssPath, css);
  console.log(`✅ ${cssPath}`);

  // Generate Tailwind Config
  console.log('📝 Generating Tailwind Config...');
  const tailwind = generateTailwindConfig();
  const tailwindPath = path.join(__dirname, '../apps/web/tailwind.config.generated.ts');
  fs.writeFileSync(tailwindPath, tailwind);
  console.log(`✅ ${tailwindPath}`);

  // Generate React Constants
  console.log('📝 Generating React Constants...');
  const reactConstants = generateReactConstants();
  const constantsPath = path.join(tsDir, 'design-tokens.ts');
  fs.writeFileSync(constantsPath, reactConstants);
  console.log(`✅ ${constantsPath}`);

  // Generate TypeScript Definitions
  console.log('📝 Generating TypeScript Definitions...');
  const typeDefs = generateTypeDefinitions();
  const typePath = path.join(typesDir, 'tokens.d.ts');
  fs.writeFileSync(typePath, typeDefs);
  console.log(`✅ ${typePath}`);

  console.log('\n✨ Design Tokens generated successfully!');
  console.log('\nNext steps:');
  console.log('1. Review generated files');
  console.log('2. Import tokens in your components');
  console.log('3. Use CSS variables or React constants');
}

main();
```

## 6. ダークモード サポート

### CSS 変数使用時

```css
/* Light mode (default) */
:root {
  --bg-primary: #FFFFFF;
  --text-primary: #212121;
}

/* Dark mode - auto detect */
@media (prefers-color-scheme: dark) {
  :root {
    --bg-primary: #121212;
    --text-primary: #FFFFFF;
  }
}

/* Dark mode - manual toggle */
.dark {
  --bg-primary: #121212;
  --text-primary: #FFFFFF;
}
```

### React コンポーネント

```typescript
export function ThemeToggle() {
  const [isDark, setIsDark] = useState(false);

  const toggleTheme = () => {
    setIsDark(!isDark);
    document.documentElement.classList.toggle('dark', !isDark);
  };

  return (
    <button onClick={toggleTheme}>
      {isDark ? '☀️ Light' : '🌙 Dark'}
    </button>
  );
}
```

## 7. Design Tokens 更新ワークフロー

### 1. Penpot でデザイン作成
   - Color palette 定義
   - Typography styles 作成
   - Component variants 追加

### 2. design-tokens.json 更新
   - Penpot の最新デザインを JSON に反映
   - WCAG コントラスト値を検証

### 3. スクリプト実行
   ```bash
   npm run tokens:generate
   ```

### 4. ファイル確認・コミット
   ```bash
   git add src/styles/tokens.css src/constants/design-tokens.ts
   git commit -m "chore: Design Tokens updated from Penpot"
   ```

### 5. React コンポーネント更新
   - 新色・新タイプを使用
   - Storybook に追加（optional）

## 8. CI/CD 統合

### GitHub Actions 例

```yaml
# .github/workflows/design-tokens.yml

name: Update Design Tokens

on:
  push:
    paths:
      - 'docs/design-tokens.json'
  workflow_dispatch:

jobs:
  generate:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-node@v3
        with:
          node-version: '18'
      
      - name: Generate Design Tokens
        run: npm run tokens:generate
      
      - name: Check if files changed
        id: verify-changed-files
        run: |
          if git diff --quiet; then
            echo "changed=false" >> $GITHUB_OUTPUT
          else
            echo "changed=true" >> $GITHUB_OUTPUT
          fi
      
      - name: Create Pull Request
        if: steps.verify-changed-files.outputs.changed == 'true'
        uses: peter-evans/create-pull-request@v4
        with:
```

## 10. ベストプラクティス

### ✅ すべき

- ✅ CSS 変数 or React Constants のいずれかを選んで統一
- ✅ Design Tokens JSON のみ編集（生成ファイル編集禁止）
- ✅ 新しい色・タイポを追加する際は常に JSON に追加

- ❌ 生成ファイル（tokens.css, design-tokens.ts）を手編集
- ❌ Penpot と JSON が不同期な状態を放置
- ❌ ハードコード色 (#0066CC など）を使用
- ❌ Tailwind config を直接編集（常に generated config を使用）
